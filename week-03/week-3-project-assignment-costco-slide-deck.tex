\section{Key Sensitivity Variables}
\begin{frame}
\frametitle{Sensitivity Variable 1: Membership Fees \textit{(Costco Wholesale Corporation, 2025)}}
\begin{itemize}
  \item Membership fees represent nearly pure profit
  \item With 3.77\% operating margin, membership fees account for \textbf{majority of operating income}
  \item A decline in renewal rates would:
    \begin{itemize}
      \item Directly reduce high-margin revenue stream
      \item Disproportionately affect profitability
    \end{itemize}
  \item \textbf{Most influential variable}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Sensitivity Variable 2: Revenue Growth \textit{(Costco Wholesale Corporation, 2026)}}
\begin{itemize}
  \item Driven by customer traffic and average transaction value
  \item Determines scale of Costco's operations
  \item Quarterly earnings releases provide trend data for investors
  \item Changes in comparable sales directly influence total revenue and profitability \textit{(Costco Wholesale Corporation, 2026)}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Sensitivity Variable 3: Operating Margin \textit{(Costco Wholesale Corporation, 2025)}}
\begin{itemize}
  \item Razor-thin operating margins (3.77\%) make company highly sensitive to cost changes
  \item Key risks:
    \begin{itemize}
      \item Tariff increases on imported goods
      \item Wage inflation
    \end{itemize}
  \item Cost structure: \$264,852M in merchandise costs and SG\&A expenses
  \item Limited room for cost fluctuations
\end{itemize}
\end{frame}

\section{Sensitivity Analysis}
\begin{frame}
\frametitle{Sensitivity Analysis Framework \textit{(Costco Wholesale Corporation, 2025)}}
\begin{itemize}
  \item Three scenarios tested:
    \begin{enumerate}
      \item \textbf{Downside Case:} Unfavorable assumptions (-5\% net sales)
      \item \textbf{Base Case:} Reasonable expectations
      \item \textbf{Upside Case:} Favorable assumptions (+5\% net sales)
    \end{enumerate}
  \item Varying net sales by ±5\%
  \item Membership fees and fixed costs held constant
\end{itemize}
\end{frame}

\section{Interpretation \& Findings}
\begin{frame}
\frametitle{Most Influential Assumption \textit{(Costco Wholesale Corporation, 2025)}}
\begin{itemize}
  \item \textbf{Membership Fee Renewal Rates} represent the most influential assumption
  \item Why?
    \begin{itemize}
      \item Membership fees flow almost entirely to operating income
      \item Small change produces disproportionately large effect on profitability
      \item Low operating margin amplifies sensitivity
    \end{itemize}
  \item \textbf{Critical Metric:} Membership retention essential for investors to monitor
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Least Influential Assumption}
\begin{itemize}
  \item \textbf{Fixed Costs} have the least impact on financial performance
  \item Why?
    \begin{itemize}
      \item Fixed costs remain constant regardless of sales volume
      \item Variations in these expenses don't significantly affect operating income
      \item Short-term perspective is most relevant
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Profitability Sensitivity}
\begin{itemize}
  \item \textbf{5\% Revenue Decrease:} EPS drops to \$17.15 (-5.9\%)
  \item \textbf{5\% Revenue Increase:} EPS rises to \$19.27 (+5.8\%)
  \item \textbf{Key Finding:} Near-linear relationship indicates proportional response to revenue changes
  \item Operating leverage of 1.16 amplifies effects slightly
  \item Moderate sensitivity to net sales changes
\end{itemize}
\end{frame}

\section{Financial Risk Assessment}
\begin{frame}
\frametitle{Risk Profile \textit{(Costco Wholesale Corporation, 2025)}}
\begin{itemize}
  \item \textbf{Overall Risk Level:} Relatively low
  \item \textbf{Why?}
    \begin{itemize}
      \item Strong membership fee revenue provides stable income stream
      \item Operating leverage of 1.16 indicates moderate sensitivity
      \item Efficient cost structure and strong cash generation
    \end{itemize}
  \item \textbf{Vulnerabilities:}
    \begin{itemize}
      \item Changes in membership renewal rates
      \item Revenue growth fluctuations
      \item Thin operating margin leaves limited buffer against adverse changes
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Variables for Monitoring}
\begin{enumerate}
  \item \textbf{Membership Renewal Rates}
    \begin{itemize}
      \item Greatest impact on financial performance
      \item Can materially affect operating income
    \end{itemize}
  \item \textbf{Revenue Growth}
    \begin{itemize}
      \item Key driver of profitability
      \item Track comparable sales trends

  \item \textbf{Cost Structure Changes}

\end{frame}

\section{Conclusion}
\begin{frame}
\frametitle{Summary}
\begin{itemize}
  \item Costco demonstrates financial resilience through stable membership fees and consistent growth
  \item 5\% revenue change produces approximately 5.9\% change in EPS
  \item Membership renewal rates are the most critical variable
  \item Fixed costs have minimal impact within normal ranges
  \item Strong membership model provides stability despite thin margins
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Investment Recommendation: HOLD}
\begin{itemize}
  \item \textbf{Strengths:}
    \begin{itemize}
      \item Financial resilience from membership fee revenue
      \item Efficient cost structure
      \item Strong cash generation
      \item Stable business model supports current valuation
    \end{itemize}
  \item \textbf{Cautions:}
    \begin{itemize}
      \item Small changes in key variables significantly impact profitability
      \item Limited upside potential without acceleration in growth or margins
      \item Sensitivity to membership renewal rates and revenue growth
    \end{itemize}
  \item \textbf{Action:} Monitor key variables closely for risks and opportunities
\end{itemize}
\end{frame}
